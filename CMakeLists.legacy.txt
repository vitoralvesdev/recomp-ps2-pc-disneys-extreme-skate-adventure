cmake_minimum_required(VERSION 3.21)
project(DisneySkatePort LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(PS2X_BUILD_RECOMP OFF CACHE BOOL "" FORCE)
set(PS2X_BUILD_ANALYZER OFF CACHE BOOL "" FORCE)
set(PS2X_BUILD_RUNTIME ON CACHE BOOL "" FORCE)
set(PS2X_BUILD_TEST OFF CACHE BOOL "" FORCE)
set(PS2X_BUILD_STUDIO OFF CACHE BOOL "" FORCE)
set(PS2X_ENABLE_DEBUG_UI OFF CACHE BOOL "" FORCE)
set(PS2X_ENABLE_AGRESSIVE_LOGS OFF CACHE BOOL "" FORCE)
set(PS2X_RUNNER_UNITY_BUILD_BATCH_SIZE 64 CACHE STRING "" FORCE)
set(DISNEY_PS2RECOMP_DIR "${CMAKE_CURRENT_SOURCE_DIR}/tools/PS2Recomp" CACHE PATH "PS2Recomp source checkout")
if(NOT EXISTS "${DISNEY_PS2RECOMP_DIR}/CMakeLists.txt")
 message(FATAL_ERROR
  "PS2Recomp source is missing at '${DISNEY_PS2RECOMP_DIR}'. The AOT runtime cannot be configured from a clean clone yet.")
endif()
add_subdirectory("${DISNEY_PS2RECOMP_DIR}" upstream)
get_target_property(runner_sources ps2EntryRunner SOURCES)
list(FILTER runner_sources EXCLUDE REGEX "register_functions[.]cpp$")
set_property(TARGET ps2EntryRunner PROPERTY SOURCES "${runner_sources}")
set(DISNEY_GUEST_BATCH_DIR "${CMAKE_CURRENT_SOURCE_DIR}/build/guest-batches" CACHE PATH "Directory containing generated guest unity batches")
file(GLOB guest_sources CONFIGURE_DEPENDS "${DISNEY_GUEST_BATCH_DIR}/*.cpp")
target_sources(ps2EntryRunner PRIVATE ${guest_sources})
target_include_directories(ps2EntryRunner BEFORE PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/src/generated-ghidra")
set_target_properties(ps2EntryRunner PROPERTIES OUTPUT_NAME DisneySkatePort RUNTIME_OUTPUT_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}/dist")
if(MSVC)
 target_compile_options(ps2EntryRunner PRIVATE /bigobj)
 target_link_options(ps2EntryRunner PRIVATE /STACK:8388608)
endif()
get_target_property(host_sources ps2EntryRunner SOURCES)
list(FILTER host_sources EXCLUDE REGEX "/src/main[.]cpp$")
set_property(TARGET ps2EntryRunner PROPERTY SOURCES "${host_sources}")
target_sources(ps2EntryRunner PRIVATE src/host/main.cpp)
get_target_property(runtime_sources ps2_runtime SOURCES)
list(FILTER runtime_sources EXCLUDE REGEX "(^|/)ps2_pad[.]cpp$")
set_property(TARGET ps2_runtime PROPERTY SOURCES "${runtime_sources}")
target_sources(ps2_runtime PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/src/host/pc_pad.cpp")


set_target_properties(ps2EntryRunner PROPERTIES UNITY_BUILD OFF)

target_sources(ps2EntryRunner PRIVATE src/host/sdk_bridges.cpp)
get_target_property(kernel_sources ps2_runtime SOURCES)
list(FILTER kernel_sources EXCLUDE REGEX "/Kernel/Syscalls/System[.]cpp$")
set_property(TARGET ps2_runtime PROPERTY SOURCES "${kernel_sources}")
target_sources(ps2_runtime PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/src/host/kernel_system.cpp")
target_sources(ps2EntryRunner PRIVATE src/host/platform_bridges.cpp)
target_sources(ps2EntryRunner PRIVATE src/host/guest_entries.cpp)
target_sources(ps2EntryRunner PRIVATE src/host/boot_diagnostics.cpp)
get_target_property(core_sources ps2_runtime SOURCES)
list(FILTER core_sources EXCLUDE REGEX "(^|/)ps2_runtime[.]cpp$")
set_property(TARGET ps2_runtime PROPERTY SOURCES "${core_sources}")
target_sources(ps2_runtime PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/src/host/pc_runtime.cpp")
add_executable(runtime_flow_test tests/runtime_flow_test.cpp)
target_link_libraries(runtime_flow_test PRIVATE ps2_runtime)
ps2x_stage_ffmpeg_runtime_dlls(runtime_flow_test)
enable_testing()
add_test(NAME runtime_flow COMMAND runtime_flow_test)
file(GLOB recovered_sources CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/src/recovered/recovered_batch_*.cpp")
target_sources(ps2EntryRunner PRIVATE ${recovered_sources} src/host/recovered_functions.cpp)

get_target_property(scheduler_sources ps2_runtime SOURCES)
list(FILTER scheduler_sources EXCLUDE REGEX "(^|/)EeScheduler[.]cpp$")
set_property(TARGET ps2_runtime PROPERTY SOURCES "${scheduler_sources}")
target_sources(ps2_runtime PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/src/host/pc_scheduler.cpp")


add_executable(graphics_driver_test tests/graphics_driver_test.cpp)
target_link_libraries(graphics_driver_test PRIVATE raylib opengl32)
add_test(NAME graphics_driver COMMAND graphics_driver_test)
